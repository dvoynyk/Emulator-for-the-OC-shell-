# ===== ТЕСТИРОВАНИЕ ЭТАПА 3: VFS =====

# Тест 1: Список файлов в корне
ls

# Тест 2: Информация о текущей директории
pwd

# Тест 3: Чтение файла из корня
cat readme.txt

# Тест 4: Переход в директорию
cd documents

# Тест 5: Список файлов в директории
ls

# Тест 6: Проверка текущего пути
pwd

# Тест 7: Возврат в корень
cd /

# Тест 8: Снова в корне
pwd

# Тест 9: Попытка перейти в несуществующую директорию (ошибка)
cd nonexistent

# Тест 10: Текущий путь все еще корень
pwd

# Тест 11: Попытка прочитать несуществующий файл (ошибка)
cat nonexistent.txt

# Тест 12: Список файлов снова
ls

# Завершение работы
exit
